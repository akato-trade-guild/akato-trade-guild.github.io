移動平均線（MA）のゴールデンクロス判定

短期MAが長期MAを下に上抜けた瞬間にtrueを返します。

// 1本前と2本前のMA値を取得
double maShort1 = iMA(Symbol(), Period(), 10, 0, MODE_SMA, PRICE_CLOSE, 1);
double maShort2 = iMA(Symbol(), Period(), 10, 0, MODE_SMA, PRICE_CLOSE, 2);
double maLong1  = iMA(Symbol(), Period(), 50, 0, MODE_SMA, PRICE_CLOSE, 1);
double maLong2  = iMA(Symbol(), Period(), 50, 0, MODE_SMA, PRICE_CLOSE, 2);

// ゴールデンクロス条件
bool isGoldenCross = (maShort2 <= maLong2) && (maShort1 > maLong1);

保有中の全ポジションを一括決済する関数
void CloseAllPositions() {
   for(int i = OrdersTotal() - 1; i >= 0; i--) {
      if(OrderSelect(i, SELECT_BY_POS, MODE_TRADES)) {
         if(OrderSymbol() == Symbol()) {
            if(OrderType() == OP_BUY) {
               bool res = OrderClose(OrderTicket(), OrderLots(), Bid, 3, clrRed);
            } else if(OrderType() == OP_SELL) {
               bool res = OrderClose(OrderTicket(), OrderLots(), Ask, 3, clrBlue);
            }
         }
      }
   }
}
